10CARD_DELIMITER = "===CARD===" 

11PAYLOAD_LOG_LIMIT = 2000 

12MAX_COMPLETE_ANSWER_CHARS = 16_000 

13# The verdict leads the answer so a refusal costs no article. Measured on the free 

14# pool with the prompt the submit box actually sends: asked as a leading judgement it 

15# withholds two coinages of six, asked as prose inside the article rules it withheld 

16# none. What it catches is nonsense strings; a well-formed compound it still cards. 

17VERDICT_PREFIX = "===USED===" 

18# The marker is looked for in the lead rather than at the very start: a model that 

19# prefixes one courtesy word must not thereby switch the judgement off. 

20VERDICT_LEAD_CHARS = 200 

21# How much may follow an opened but unclosed judgement before it is given up on. A 

22# judgement wrapped over several lines closes well inside this even with a wordy 

23# register; one that never closes would otherwise hold the whole article back for 

24# good. Set too tight, the bound expires mid-object and prints the rest of the JSON 

25# to the reader as prose. 

26MAX_VERDICT_JSON_CHARS = 400

30_PROMPT = """You are a language tutor. The request below concerns {source_lang}. 

31 

32{request} 

33 

34WRITE YOUR ENTIRE ANSWER IN {target_lang}. Explanations, translations, sense 

35labels, usage notes, origin and reasons are in {target_lang}. Only a headword, 

36quoted source text and source-language examples stay in {source_lang}. 

37{source_hints} 

38 

39First decide whether the submission is ONE LEXICAL UNIT WORTH LEARNING WHOLE or 

40text containing units. A single source word is a unit, and so is a multi-word 

41expression whose whole wording is the reusable lookup target: an idiom, 

42collocation, phrasal or separable verb or conventional formula. Its dictionary 

43form may differ from the submitted one and its pieces may stand apart. Anything 

44that reports a particular situation is text, even when a fixed expression fills 

45most of it — return that expression separately in combinations rather than making 

46its changing context part of a dictionary entry. If uncertain, choose text. 

47{intent_rule} 

48 

49Having chosen the unit branch, judge the submitted wording before writing about 

50it, and open the answer with that judgement on its own line: 

51{verdict_prefix} {{"used": true or false, "where": "<the register, field, dialect or 

52period this exact wording is used in, in a few words; empty when used is false>"}} 

53Rarity is no objection: wording real speakers use in any register, field, dialect 

54or period is used, however uncommon. Wording that is merely well formed — a 

55compound, derivation or coinage nobody actually says — is not used. A misspelling 

56is not this case: it is real wording written wrong, so answer it as the misspelling 

57rules below require, with the correction in suggestion. Say used false only when 

58nothing attested is close at all, and then write nothing else: no article, no card, 

59no explanation. The text branch has no such line. 

60 

61For a unit, write a complete compact dictionary article in this order: 

621. Begin with the unit heading in <b> tags, using the dictionary lemma, which is 

63 also what goes in word; head a suspected misspelling with the correction, never 

64 with the submitted spelling. Before writing anything else, check whether changing 

65 a letter or two of a correctly spelled submission spells a markedly commoner word 

66 — quiet beside quite, Rate beside Ratte. When it does, keep the heading on the 

67 submission and name that commoner word in suggestion. 

682. Give the translations, most frequent in everyday speech first. Never name the 

69 part of speech. Put a register mark after the translation it qualifies. 

703. Forms only when useful to recognise or produce. Use a table of at most six 

71 rows, each with a short {source_lang} form or phrase and its {target_lang} 

72 rendering. Skip forms completely for invariable words and fixed expressions. 

734. Usage: collocations, governed prepositions, confusions, register and 

74 countability where relevant. 

755. Origin: always include it; 1-3 sentences for a borrowing, one compact line for 

76 a native word. 

776. Give 2-4 short everyday examples with translations. 

78For a set expression, also explain what its parts contribute and why the whole 

79means what it does. With supplied context, lead with the sense used there but 

80keep all other senses below it. 

81 

82For text, begin with a natural translation of the whole submitted text. Then give 

832-5 compact notes about genuinely difficult constructions, word order, forms, 

84set expressions or meanings. Do not write a dictionary entry about one selected 

85word. 

86 

87No phonetic transcription. Use only <b>, <i>, <table>, <tr>, and <td>, with no 

88attributes; tables are only for forms. No Markdown emphasis, headings, code 

89fences, preamble or closing remarks. The article is at most 3500 characters. 

90 

91After the article output ===CARD=== on its own line, followed immediately by one 

92line of JSON matching ONE of these neutral schematic branches. Angle-bracketed 

93values are placeholders, not strings to copy: 

94 

95{{"kind": "unit", "word": "<dictionary lemma of the unit>", 

96 "word_relation": "<same, morphology or typo>", 

97 "suggestion": "<corrected spelling, likelier word, or empty>", 

98 "meanings": [{{"label": "<short target-language sense label or empty>", 

99 "translations": ["<target-language translation>"], 

100 "examples": [{{"highlighted": "<short source-language sentence, unit in b tags>", 

101 "translation": "<target-language translation>"}}]}}], 

102 "segments": [{{"label": "<component dictionary form>", 

103 "surface": "<component form seen in the expression>", 

104 "why": "<short target-language reason>"}}]{context_field}}} 

105 

106OR 

107 

108{{"kind": "text", "combinations": [{{"label": "<dictionary form of the unit>", 

109 "surface": "<the same unit as the text spells it>", 

110 "why": "<short target-language reason>"}}]}} 

111 

112word_relation is typo when the submission is misspelled, and then suggestion holds 

113the correct spelling; morphology when word is a different dictionary form of a 

114correctly spelled submission; same when word is the submission itself. 

115 

116suggestion also holds the markedly commoner near-spelling of rule 1, and the relation 

117then stays same or morphology: that submission is real, so the article and card stay 

118about it and the commoner word is only offered. Never offer a synonym, or a commoner 

119word spelled differently. suggestion is empty otherwise. 

120 

121meanings are the senses that need different words in {target_lang}, most common 

122first; do not impose a numerical limit. Every meaning has 2-4 main translations 

123and 1-2 examples. When several meanings remain, every label is a short 

124{target_lang} tag distinguishing them; for one meaning its label is empty. 

125Each highlighted example is a whole sentence carrying <b> tags around all and 

126only the unit, since it becomes the front of a card. Mark a contiguous unit with 

127one span and separated or reflexive pieces with one span each, in their original 

128positions. Never mark a subject, object, auxiliary or argument merely because it 

129occurs with the unit, and always leave at least one unmarked source-language word 

130in the sentence. {context_rule} 

131 

132For a multi-word set expression, segments contains every word-shaped component, 

133including particles and prepositions, with no count cap. Preserve the forms seen 

134in the submitted expression. Do not repeat the whole expression as a component. 

135 

136For text, put every clear multi-word lookup target in combinations, even when one 

137accounts for most of the utterance. It qualifies when its meaning does not follow 

138from its words one at a time, or when its lexical pieces stand apart so no piece 

139can be looked up alone: a separable or reflexive verb, governed combination, 

140collocation or set expression. A single ordinary word and an arbitrary tense, 

141negation or current argument do not qualify. Keep distinct non-overlapping units 

142separate. Return every clear unit, do not pad the list, and use an empty list when 

143nothing qualifies. 

144 

145label and surface are the same unit twice: label is its dictionary form, and 

146surface holds its lexical pieces copied token for token out of the submitted 

147text, in the same spelling, script and capitalization and in source order, with 

148an ellipsis joining pieces which stand apart. So label may read `aufstehen` 

149while surface reads `steht ... auf`. Copy surface out of the sentence: never 

150translate, transliterate, correct or lemmatise it. Include every fixed piece — 

151reflexive particle, separable particle, governed preposition, support verb — in 

152the form it takes in this sentence, and leave out negation and the current 

153subject, object or complement. Do not enumerate ordinary words: the backend 

154gives every source word its own chip anyway. Text JSON contains no word, 

155meanings, segments or other card fields. 

156 

157Check before answering that the JSON is valid, with every string in double 

158quotes, and that a unit article begins with the bold heading its word names. 

159"""

199def build_prompt( # noqa: PLR0913 - one prompt, and every switch that varies it. 

200 language: Language, 

201 word: str, 

202 target_lang: str, 

203 *, 

204 context: str = "", 

205 unit_intent: bool = False, 

206) -> str: 

207 """Build the sole compact prompt; only the request intent varies.""" 

208 request = ( 

209 f'Make a card for this selected unit: "{word}"\nContext: "{context}"' 

210 if unit_intent and context 

211 else f'Make a card for this selected unit: "{word}"' 

212 if unit_intent 

213 else f'Analyse this submitted text: "{word}"' 

214 ) 

215 context_field = ', "context_sense": <zero-based index>' if context else "" 

216 intent_rule = _UNIT_INTENT if unit_intent else _OPEN_INTENT 

217 return _PROMPT.format( 

218 source_lang=language.name, 

219 target_lang=target_lang, 

220 source_hints=language.prompt_hints or "", 

221 request=request, 

222 verdict_prefix=VERDICT_PREFIX, 

223 intent_rule=intent_rule, 

224 context_field=context_field, 

225 context_rule=_CONTEXT_RULE if context else _NO_CONTEXT_RULE, 

226 )

246@dataclass(frozen=True) 

247class Verdict: 

248 """The answer's own judgement on whether the submitted wording is used at all. 

249 

250 The prompt also asks where it is used: naming a register or period is what makes 

251 the judgement concrete rather than a guess. Nothing here reads that answer back, 

252 so it is not kept. 

253 """ 

254 

255 used: bool

258def _marker_at(raw: str) -> int | None: 

259 """Where the verdict marker starts in the answer's lead, whole or still arriving.""" 

260 # The window covers a marker that *starts* inside the lead, so one beginning at its 

261 # last character is still found whole rather than read as prose. 

262 lead = raw[: VERDICT_LEAD_CHARS + len(VERDICT_PREFIX)] 

263 at = lead.find(VERDICT_PREFIX) 

264 if 0 <= at < VERDICT_LEAD_CHARS: 

265 return at 

266 # A marker still arriving may start anywhere the whole one may, so the two scans 

267 # share a window: gating the partial one tighter showed half a marker as prose. 

268 if len(raw) > VERDICT_LEAD_CHARS + len(VERDICT_PREFIX): 

269 return None 

270 # Only the end of the stream can hold half a marker, and it must not flash on 

271 # the page as prose before the judgement it belongs to is readable. 

272 for length in range(min(len(VERDICT_PREFIX) - 1, len(raw)), 0, -1): 

273 if raw.endswith(VERDICT_PREFIX[:length]) and len(raw) - length < VERDICT_LEAD_CHARS: 

274 return len(raw) - length 

275 return None

278def _read_verdict(raw: str) -> tuple[Verdict | None, int, int | None] | None: 

279 """Scan the lead for the verdict: what it says, where it starts, where it ends. 

280 

281 None when no marker has arrived at all. An end of None means the marker is still 

282 arriving and everything from it is held back; a marker that has arrived whole but 

283 cannot be read yields no verdict and is dropped, because unreadable is absent and 

284 the marker itself is not prose the reader is owed. 

285 """ 

286 at = _marker_at(raw) 

287 if at is None: 

288 return None 

289 after = at + len(VERDICT_PREFIX) 

290 tail = raw[after:] 

291 indent = len(tail) - len(tail.lstrip()) 

292 try: 

293 value, consumed = json.JSONDecoder().raw_decode(tail[indent:]) 

294 except ValueError: 

295 opened = tail.lstrip().startswith("{") 

296 if opened and len(tail) <= MAX_VERDICT_JSON_CHARS: 

297 # A judgement written over more than one line has not finished arriving. 

298 # Cutting it at the first newline would print the rest of its JSON as prose. 

299 return (None, at, None) 

300 line_end = tail.find("\n") 

301 # No line break yet means the judgement may still be on its way. 

302 return (None, at, None if line_end < 0 else after + line_end + 1) 

303 end = after + indent + consumed 

304 used = value.get("used") if isinstance(value, dict) else None 

305 verdict = Verdict(used) if isinstance(used, bool) else None 

306 return verdict, at, end + len(raw[end:]) - len(raw[end:].lstrip())

318def parse_attestation(raw: str) -> Verdict | None: 

319 """Read the standalone answer, which is one bare JSON object and no marker.""" 

320 start = raw.find("{") 

321 if start < 0: 

322 return None 

323 try: 

324 value, _consumed = json.JSONDecoder().raw_decode(raw[start:]) 

325 except ValueError: 

326 return None 

327 used = value.get("used") if isinstance(value, dict) else None 

328 return Verdict(used) if isinstance(used, bool) else None

331def parse_verdict(raw: str) -> Verdict | None: 

332 """Read the answer's verdict, or None when it gave none this side of readable.""" 

333 read = _read_verdict(raw) 

334 return None if read is None else read[0]

337def strip_verdict(raw: str) -> str: 

338 """Drop the verdict from what the reader sees, and every partial prefix of it.""" 

339 read = _read_verdict(raw) 

340 if read is None: 

341 return raw 

342 _, at, end = read 

343 return raw[:at] if end is None else raw[:at] + raw[end:]

346def extract_answer( 

347 raw: str, 

348 submitted: str, 

349 language: Language, 

350 *, 

351 unit_intent: bool = False, 

352 context: str = "", 

353) -> ParsedAnswer | None: 

354 """Extract and validate the hidden answer, returning None when unusable.""" 

355 if len(raw) > MAX_COMPLETE_ANSWER_CHARS: 

356 logger.warning( 

357 "oversized answer for %s/%r: %s characters exceeds the %s-character bound", 

358 language.code, 

359 submitted, 

360 len(raw), 

361 MAX_COMPLETE_ANSWER_CHARS, 

362 ) 

363 return None 

364 delimiter_at = raw.find(CARD_DELIMITER) 

365 if delimiter_at < 0: 

366 logger.warning( 

367 "no answer block for %s/%r: the answer never wrote the delimiter", 

368 language.code, 

369 submitted, 

370 ) 

371 return None 

372 payload = raw[delimiter_at + len(CARD_DELIMITER) :] 

373 try: 

374 return parse_answer_payload( 

375 payload, 

376 submitted, 

377 language, 

378 unit_intent=unit_intent, 

379 context=context, 

380 ) 

381 except CardParseError as exc: 

382 logger.warning( 

383 "unusable answer block for %s/%r: %s; payload %s", 

384 language.code, 

385 submitted, 

386 exc, 

387 _logged(payload), 

388 ) 

389 return None

392def _logged(payload: str) -> str: 

393 payload = payload.strip() 

394 if len(payload) <= PAYLOAD_LOG_LIMIT: 

395 return repr(payload) 

396 return f"{payload[:PAYLOAD_LOG_LIMIT]!r} … ({len(payload)} chars)"
