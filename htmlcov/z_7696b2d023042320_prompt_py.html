1"""The card-producing vocabulary prompt.""" 

2 

3from echo_words.languages import Language 

4 

5_PROMPT = """You are a vocabulary tutor. The word below is in {source_lang}. Analyse 

6this word or short phrase: {word} 

7 

8WRITE YOUR ENTIRE ANSWER IN {target_lang}. 

9This is absolute. Every part of the answer — the translations, the labels, 

10the register marks, the usage notes, the origin, the explanations of the 

11examples — is written in {target_lang}, and in no other language. These 

12instructions are in English; that says nothing about the answer language. 

13Do not answer in English unless {target_lang} IS English. Do not answer in 

14{source_lang} unless {target_lang} IS {source_lang}. Two things, and only 

15these two, stay in {source_lang}: the headword itself and the example 

16sentences (each of which is followed by its {target_lang} translation). 

17 

18{context_note}Be compact. No preamble, no closing remarks. {source_hints} 

19The order of the sections is fixed: 

20 

211. First line: the word being analysed, in bold. 

222. Translations, most frequent in everyday speech first; each with its 

23 part of speech and a register mark (colloquial, formal, slang, vulgar 

24 and so on) where it matters. 

253. Usage: typical collocations and prepositions, what it is confused 

26 with; countability and irregular forms where they matter. 

274. Origin: if the word was borrowed, 1-3 sentences on which language it 

28 came from and how it travelled; for a native word, one line. 

295. Examples: 2-4 short everyday sentences, each followed by its translation. 

30 

31Give no phonetic transcription: pronunciation is delivered as audio. 

32 

33Analyse EXACTLY the word given and do NOT substitute another. If it looks 

34like a typo, do not silently fix it: add one short line beginning with ✏️ 

35and naming the likely intended spelling, and put that spelling in the 

36suggestion field of the card JSON. If there is no typo, suggestion is an 

37empty string. 

38If it is an idiom or a phrasal verb, explain both the literal and the 

39figurative sense and when it is used. 

40ALL EMPHASIS IS HTML. Use exactly two tags and no others: <b> around the 

41headword, <i> around the {source_lang} example sentences and any 

42{source_lang} form quoted inside the text. Emphasis punctuation of any 

43kind is forbidden — no markdown, no headings, no bullet markers, no 

44tables, no code fences. 

45The whole analysis: at most 3500 characters. 

46 

47After the analysis output the line ===CARD=== exactly, and immediately 

48after it one line of JSON, with no commentary and no HTML tags inside the 

49values: 

50{{"word": "...", "suggestion": "...", 

51 "meanings": [{{"label": "...", "pos": "...", "translations": ["...", "..."], 

52 "examples": [{{"text": "...", "translation": "..."}}]}}]}} 

53word — the input word exactly as given (never corrected); suggestion — the 

54likely intended spelling of a typo, or an empty string. 

55meanings normally holds one element with an empty label. Split it into 

56several (at most three) only when the senses are genuinely unrelated (like 

57bank "financial institution" and bank "river edge"); then label is a 

581-3 word tag in {target_lang} telling the senses apart. pos is that 

59sense's part of speech as one short abbreviation in {target_lang}. 

60translations — the 2-4 main {target_lang} translations of that sense; 

61examples — the 1-2 shortest examples of that very sense: text is a 

62sentence in {source_lang}, translation is its rendering in {target_lang}. 

63In at least one example per sense, use the headword in exactly the form it 

64was given, unless that makes the sentence unnatural. 

65 

66Before you answer, check two things once more. Is every word of it in 

67{target_lang}, except the headword and the {source_lang} example sentences? 

68And is every emphasis an HTML tag, with no punctuation used for emphasis 

69anywhere? 

70""" 

71 

72_CONTEXT_NOTE = """The word was met in this context: "{context}" 

73Analyse the sense in which it is used there. If that sense is rare, 

74domain-specific or not recorded in dictionaries at all, say so plainly and 

75explain that sense — do not substitute the nearest dictionary meaning. 

76 

77""" 

78 

79 

80def build_prompt(language: Language, word: str, target_lang: str, *, context: str = "") -> str: 

81 """Build the compact analysis prompt; card parsing is added in M4.""" 

82 context_note = _CONTEXT_NOTE.format(context=context) if context else "" 

83 return _PROMPT.format( 

84 source_lang=language.name, 

85 target_lang=target_lang, 

86 source_hints=language.prompt_hints or "", 

87 word=word, 

88 context_note=context_note, 

89 )
