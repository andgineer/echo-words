8CARD_DELIMITER = "===CARD===" 

9PAYLOAD_LOG_LIMIT = 2000 

10MAX_COMPLETE_ANSWER_CHARS = 16_000 

11 

12logger = logging.getLogger(__name__)

14_PROMPT = """You are a language tutor. The request below concerns {source_lang}. 

15 

16{request} 

17 

18WRITE YOUR ENTIRE ANSWER IN {target_lang}. Explanations, translations, sense 

19labels, usage notes, origin and reasons are in {target_lang}. Only a headword, 

20quoted source text and source-language examples stay in {source_lang}. 

21{source_hints} 

22 

23First decide whether the submission is ONE LEXICAL UNIT WORTH LEARNING WHOLE or 

24text containing units. A single source word is a unit, and so is a multi-word 

25expression whose whole wording is the reusable lookup target: an idiom, 

26collocation, phrasal or separable verb or conventional formula. Its dictionary 

27form may differ from the submitted one and its pieces may stand apart. Anything 

28that reports a particular situation is text, even when a fixed expression fills 

29most of it — return that expression separately in combinations rather than making 

30its changing context part of a dictionary entry. If uncertain, choose text. 

31{intent_rule} 

32 

33For a unit, write a complete compact dictionary article in this order: 

341. Begin with the unit heading in <b> tags, using the dictionary lemma; when you 

35 suspect a misspelling, head the article with the submitted spelling instead. 

362. Give the translations, most frequent in everyday speech first. Never name the 

37 part of speech. Put a register mark after the translation it qualifies. 

383. Forms only when useful to recognise or produce. Use a table of at most six 

39 rows, each with a short {source_lang} form or phrase and its {target_lang} 

40 rendering. Skip forms completely for invariable words and fixed expressions. 

414. Usage: collocations, governed prepositions, confusions, register and 

42 countability where relevant. 

435. Origin: always include it; 1-3 sentences for a borrowing, one compact line for 

44 a native word. 

456. Give 2-4 short everyday examples with translations. 

46For a set expression, also explain what its parts contribute and why the whole 

47means what it does. With supplied context, lead with the sense used there but 

48keep all other senses below it. 

49 

50For text, begin with a natural translation of the whole submitted text. Then give 

512-5 compact notes about genuinely difficult constructions, word order, forms, 

52set expressions or meanings. Do not write a dictionary entry about one selected 

53word. 

54 

55No phonetic transcription. Use only <b>, <i>, <table>, <tr>, and <td>, with no 

56attributes; tables are only for forms. No Markdown emphasis, headings, code 

57fences, preamble or closing remarks. The article is at most 3500 characters. 

58 

59After the article output ===CARD=== on its own line, followed immediately by one 

60line of JSON matching ONE of these neutral schematic branches. Angle-bracketed 

61values are placeholders, not strings to copy: 

62 

63{{"kind": "unit", "word": "<dictionary lemma of the unit>", 

64 "word_relation": "<same, morphology or typo>", 

65 "suggestion": "<corrected spelling, or empty>", 

66 "meanings": [{{"label": "<short target-language sense label or empty>", 

67 "translations": ["<target-language translation>"], 

68 "examples": [{{"highlighted": "<short source-language sentence, unit in b tags>", 

69 "translation": "<target-language translation>"}}]}}], 

70 "segments": [{{"label": "<component dictionary form>", 

71 "surface": "<component form seen in the expression>", 

72 "why": "<short target-language reason>"}}]{context_field}}} 

73 

74OR 

75 

76{{"kind": "text", "combinations": [{{"label": "<dictionary form of the unit>", 

77 "surface": "<the same unit as the text spells it>", 

78 "why": "<short target-language reason>"}}]}} 

79 

80word_relation is typo when the submission is misspelled, and then suggestion holds 

81the correct spelling; morphology when word is a different dictionary form of a 

82correctly spelled submission; same when word is the submission itself. suggestion 

83is empty unless the relation is typo. 

84 

85meanings are the senses that need different words in {target_lang}, most common 

86first; do not impose a numerical limit. Every meaning has 2-4 main translations 

87and 1-2 examples. When several meanings remain, every label is a short 

88{target_lang} tag distinguishing them; for one meaning its label is empty. 

89Each highlighted example is a whole sentence carrying <b> tags around all and 

90only the unit, since it becomes the front of a card. Mark a contiguous unit with 

91one span and separated or reflexive pieces with one span each, in their original 

92positions. Never mark a subject, object, auxiliary or argument merely because it 

93occurs with the unit, and always leave at least one unmarked source-language word 

94in the sentence. {context_rule} 

95 

96For a multi-word set expression, segments contains every word-shaped component, 

97including particles and prepositions, with no count cap. Preserve the forms seen 

98in the submitted expression. Do not repeat the whole expression as a component. 

99 

100For text, put every clear multi-word lookup target in combinations, even when one 

101accounts for most of the utterance. It qualifies when its meaning does not follow 

102from its words one at a time, or when its lexical pieces stand apart so no piece 

103can be looked up alone: a separable or reflexive verb, governed combination, 

104collocation or set expression. A single ordinary word and an arbitrary tense, 

105negation or current argument do not qualify. Keep distinct non-overlapping units 

106separate. Return every clear unit, do not pad the list, and use an empty list when 

107nothing qualifies. 

108 

109label and surface are the same unit twice: label is its dictionary form, and 

110surface holds its lexical pieces copied token for token out of the submitted 

111text, in the same spelling, script and capitalization and in source order, with 

112an ellipsis joining pieces which stand apart. So label may read `aufstehen` 

113while surface reads `steht ... auf`. Copy surface out of the sentence: never 

114translate, transliterate, correct or lemmatise it. Include every fixed piece — 

115reflexive particle, separable particle, governed preposition, support verb — in 

116the form it takes in this sentence, and leave out negation and the current 

117subject, object or complement. Do not enumerate ordinary words: the backend 

118gives every source word its own chip anyway. Text JSON contains no word, 

119meanings, segments or other card fields. 

120 

121Check before answering that the JSON is valid, with every string in double 

122quotes, and that a unit article begins with the bold heading its word names. 

123"""

125_UNIT_INTENT = "The learner explicitly selected a unit, so kind must be unit." 

126_OPEN_INTENT = "For this submit-box request, choose the branch yourself." 

127# Appended to the intent line rather than placed on its own, so the prompt every 

128# other request sends stays byte for byte what the benchmark measured. 

129_CONFIRMED_SPELLING = ( 

130 "The learner has confirmed this spelling deliberately, so treat it as correct: " 

131 "analyse exactly the submitted spelling, copy it into word unchanged, set " 

132 "word_relation to same, leave suggestion empty, and spell it that way in every " 

133 "example and heading." 

134) 

135_CONTEXT_RULE = ( 

136 "Use the supplied context as the first example of its sense and include " 

137 '"context_sense": <zero-based index> in the unit object. In that example, ' 

138 "copy the submitted selected-unit surface exactly, mark those selected " 

139 "tokens and no others, and do not expand the selection to neighbouring context." 

140) 

141_NO_CONTEXT_RULE = "Do not add a field selecting a contextual sense." 

142 

143_EXTENDED_PROMPT = """You are a lexicographer. The word below is in {source_lang}. 

144Analyse it in depth: {word} 

145 

146WRITE YOUR ENTIRE ANSWER IN {target_lang}, and in no other language. Only the 

147headword and example sentences stay in {source_lang}, and every example is 

148followed by its {target_lang} translation. 

149 

150{context_note}The reader has already seen the short entry. Cover every sense, 

151including domain-specific, archaic, regional and slang senses; origin and its 

152route in depth; usage and common mistakes; shades and near-synonyms; and one or 

153two examples per sense. Use only <b> and <i>, no Markdown. Give no phonetic 

154transcription, JSON or delimiters. 

155"""

158def build_prompt( # noqa: PLR0913 - one prompt, and every switch that varies it. 

159 language: Language, 

160 word: str, 

161 target_lang: str, 

162 *, 

163 context: str = "", 

164 unit_intent: bool = False, 

165 spelling_confirmed: bool = False, 

166) -> str: 

167 """Build the sole compact prompt; only the request intent varies.""" 

168 request = ( 

169 f'Make a card for this selected unit: "{word}"\nContext: "{context}"' 

170 if unit_intent and context 

171 else f'Make a card for this selected unit: "{word}"' 

172 if unit_intent 

173 else f'Analyse this submitted text: "{word}"' 

174 ) 

175 context_field = ', "context_sense": <zero-based index>' if context else "" 

176 intent_rule = _UNIT_INTENT if unit_intent else _OPEN_INTENT 

177 if spelling_confirmed: 

178 intent_rule = f"{intent_rule} {_CONFIRMED_SPELLING}" 

179 return _PROMPT.format( 

180 source_lang=language.name, 

181 target_lang=target_lang, 

182 source_hints=language.prompt_hints or "", 

183 request=request, 

184 intent_rule=intent_rule, 

185 context_field=context_field, 

186 context_rule=_CONTEXT_RULE if context else _NO_CONTEXT_RULE, 

187 )

207def extract_answer( 

208 raw: str, 

209 submitted: str, 

210 language: Language, 

211 *, 

212 unit_intent: bool = False, 

213 context: str = "", 

214) -> ParsedAnswer | None: 

215 """Extract and validate the hidden answer, returning None when unusable.""" 

216 if len(raw) > MAX_COMPLETE_ANSWER_CHARS: 

217 logger.warning( 

218 "oversized answer for %s/%r: %s characters exceeds the %s-character bound", 

219 language.code, 

220 submitted, 

221 len(raw), 

222 MAX_COMPLETE_ANSWER_CHARS, 

223 ) 

224 return None 

225 delimiter_at = raw.find(CARD_DELIMITER) 

226 if delimiter_at < 0: 

227 logger.warning( 

228 "no answer block for %s/%r: the answer never wrote the delimiter", 

229 language.code, 

230 submitted, 

231 ) 

232 return None 

233 payload = raw[delimiter_at + len(CARD_DELIMITER) :] 

234 try: 

235 return parse_answer_payload( 

236 payload, 

237 submitted, 

238 language, 

239 unit_intent=unit_intent, 

240 context=context, 

241 ) 

242 except CardParseError as exc: 

243 logger.warning( 

244 "unusable answer block for %s/%r: %s; payload %s", 

245 language.code, 

246 submitted, 

247 exc, 

248 _logged(payload), 

249 ) 

250 return None

253def _logged(payload: str) -> str: 

254 payload = payload.strip() 

255 if len(payload) <= PAYLOAD_LOG_LIMIT: 

256 return repr(payload) 

257 return f"{payload[:PAYLOAD_LOG_LIMIT]!r} … ({len(payload)} chars)"
