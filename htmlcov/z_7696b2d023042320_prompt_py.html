508def reverse_example_issue( 

509 example: str, 

510 form: str, 

511 word: str, 

512 language: Language, 

513 target: str = DEFAULT_TARGET_LANGUAGE, 

514) -> str | None: 

515 """Why the card could not use this sentence as its front, or None when it could. 

516 

517 It is the card's own test: a sentence the card cannot mark the word in would fail 

518 the card later, and a whole-word comparison would refuse every inflected example. 

519 """ 

520 if not example: 

521 return "missing" 

522 if len(example) > MAX_CONTEXT_LENGTH: 

523 return "too_long" 

524 if validate_text(example, language) is not None: 

525 return "script" 

526 if not sentence_is_source_language(example, language, target): 

527 return "letters" 

528 related = bool(form) and _spells_the_word(form, word, language) 

529 marked = context_sentence_forms(example, form, language, target) if related else None 

530 if marked is None: 

531 marked = context_sentence_forms(example, word, language, target) 

532 if marked is not None: 

533 return None 

534 return "unrelated" if form and not related else "form"

537def _headword_case(word: str, form: str, example: str) -> str: 

538 """The headword in the letter case its own sentence writes it in, where that shows. 

539 

540 Asked for a word as a dictionary heads it, answers come back in dictionary 

541 typography — all capitals, or every headword capitalised — and with German nouns 

542 lowered; a single word standing mid-sentence shows the case the language gives it. 

543 """ 

544 if word.isupper() and len(word) > 1: 

545 word = word.lower() 

546 if len(word.split()) > 1: 

547 return word 

548 tokens = [match.group() for match in _WORD_TOKEN.finditer(example)] 

549 wanted = (form or word).casefold() 

550 position = next( 

551 (index for index, token in enumerate(tokens) if token.casefold() == wanted), 

552 None, 

553 ) 

554 if position is None: 

555 return word 

556 if position == 0: 

557 # Opening the sentence hides the case. A form the answer copied in lower case 

558 # shows it; one in capitals may only have copied the sentence's first letter. 

559 lowered = form[:1].islower() and form.casefold() == word.casefold() 

560 return form if lowered else word 

561 spelled = tokens[position] 

562 if spelled.casefold() == word.casefold(): 

563 return spelled 

564 first = word[0].upper() if spelled[0].isupper() else word[0].lower() 

565 return first + word[1:]

567 

568def _spells_the_word(form: str, word: str, language: Language) -> bool: 

569 """Whether every token of the form is a form of some token of the word. 

570 

571 A sentence written around another word would be marked at that word, so the form 

572 has to share a stem with its word; only suppletive forms (went for go) fail this. 

573 """ 

574 stems = [_stem_fold(token, language) for token in _WORD_TOKEN.findall(word)] 

575 return bool(stems) and all( 

576 any(_shares_a_stem(_stem_fold(token, language), stem) for stem in stems) 

577 for token in _WORD_TOKEN.findall(form) 

578 ) 

579 

580 

581def _stem_fold(token: str, language: Language) -> str: 

582 decomposed = unicodedata.normalize("NFD", fold_for_match(token, language)) 

583 return "".join(char for char in decomposed if not unicodedata.combining(char)) 

584 

585 

586def _shares_a_stem(token: str, stem: str) -> bool: 

587 # Compared from any point of the word's token, so a separated particle (stehe ... 

588 # auf for aufstehen) matches; a vowel change of one letter (gave, give) passes too. 

589 shorter = min(len(token), len(stem)) 

590 wanted = min(max(3, shorter - 2), shorter) 

591 longest = max(len(os.path.commonprefix((token, stem[offset:]))) for offset in range(len(stem))) 

592 return longest >= wanted or _within_one_edit(token, stem) 

593 

594 

595def _within_one_edit(first: str, second: str) -> bool: 

596 if abs(len(first) - len(second)) > 1: 

597 return False 

598 if len(first) == len(second): 

599 return sum(a != b for a, b in zip(first, second, strict=True)) <= 1 

600 longer, shorter = (first, second) if len(first) > len(second) else (second, first) 

601 return any(longer[:index] + longer[index + 1 :] == shorter for index in range(len(longer)))

604def _equivalents(value: Any, language: Language, target: str) -> tuple[Equivalent, ...]: 

605 if not isinstance(value, list): 

606 return () 

607 kept: list[Equivalent] = [] 

608 seen: set[str] = set() 

609 for item in value: 

610 if not isinstance(item, dict): 

611 continue 

612 word = plain_unit(_plain(item.get("word"))) 

613 folded = fold_for_match(word, language) 

614 if validate_word(word, language) is not None or folded in seen: 

615 continue 

616 seen.add(folded) 

617 example, form = _plain(item.get("example")), _plain(item.get("form")) 

618 if reverse_example_issue(example, form, word, language, target) is None: 

619 kept.append(Equivalent(_headword_case(word, form, example), example)) 

620 else: 

621 kept.append(Equivalent(_headword_case(word, "", ""), "")) 

622 if len(kept) == MAX_EQUIVALENTS: 

623 break 

624 return tuple(kept)

627def _plain(value: Any) -> str: 

628 if not isinstance(value, str): 

629 return "" 

630 return " ".join(unicodedata.normalize("NFC", value).split())

633def json_object(raw: str) -> dict | None: 

634 """The JSON object the text opens with, read past a string value missing its opening quote.""" 

635 start = raw.find("{") 

636 if start < 0: 

637 return None 

638 text = raw[start:] 

639 for candidate in (text, _UNQUOTED_VALUE.sub(r'\1"', text)): 

640 try: 

641 value, _consumed = json.JSONDecoder().raw_decode(candidate) 

642 except ValueError: 

643 continue 

644 return value if isinstance(value, dict) else None 

645 return None

648def extract_answer( # noqa: PLR0913 - the whole request the answer is read against. 

649 raw: str, 

650 submitted: str, 

651 language: Language, 

652 *, 

653 unit_intent: bool = False, 

654 context: str = "", 

655 target: str = DEFAULT_TARGET_LANGUAGE, 

656) -> ParsedAnswer | None: 

657 """Extract and validate the hidden answer, returning None when unusable.""" 

658 if len(raw) > MAX_COMPLETE_ANSWER_CHARS: 

659 logger.warning( 

660 "oversized answer for %s/%r: %s characters exceeds the %s-character bound", 

661 language.code, 

662 submitted, 

663 len(raw), 

664 MAX_COMPLETE_ANSWER_CHARS, 

665 ) 

666 return None 

667 delimiter_at = raw.find(CARD_DELIMITER) 

668 if delimiter_at < 0: 

669 logger.warning( 

670 "no answer block for %s/%r: the answer never wrote the delimiter", 

671 language.code, 

672 submitted, 

673 ) 

674 return None 

675 payload = raw[delimiter_at + len(CARD_DELIMITER) :] 

676 try: 

677 return parse_answer_payload( 

678 payload, 

679 submitted, 

680 language, 

681 unit_intent=unit_intent, 

682 context=context, 

683 target=target, 

684 ) 

685 except CardParseError as exc: 

686 # The context is logged beside the payload because most rejections are about 

687 # the two of them disagreeing: without the sentence that was asked about, a 

688 # copy that missed it by one word reads exactly like an invented one. 

689 logger.warning( 

690 "unusable answer block for %s/%r in context %r: %s; payload %s", 

691 language.code, 

692 submitted, 

693 context, 

694 exc, 

695 _logged(payload), 

696 ) 

697 return None
