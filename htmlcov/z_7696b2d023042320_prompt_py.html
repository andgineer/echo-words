30_BRANCH = """First decide whether the submission is ONE LEXICAL UNIT WORTH LEARNING WHOLE or 

31text containing units. A single source word is a unit, and so is a multi-word 

32expression whose whole wording is the reusable lookup target: an idiom, 

33collocation, phrasal or separable verb or conventional formula. Its dictionary 

34form may differ from the submitted one and its pieces may stand apart. Anything 

35that reports a particular situation is text, even when a fixed expression fills 

36most of it — return that expression separately in combinations rather than making 

37its changing context part of a dictionary entry. A clause with its own subject and 

38finite verb reports a situation however ordinary a thing it is to say, so it is

41_ARTICLE_RULES = """1. Begin with the unit heading in <b> tags, using the dictionary lemma, which is 

42 also what goes in word; head a suspected misspelling with the correction, never 

43 with the submitted spelling. 

442. Give the translations, most frequent in everyday speech first. Never name the 

45 part of speech. Put a register mark after the translation it qualifies. 

463. Forms only when useful to recognise or produce. Use a table of at most six 

47 rows, each with a short {source_lang} form or phrase and its {target_lang} 

48 rendering. Name no grammatical category in it — no case, tense, person, 

49 number, gender or part of speech — because the phrase carries the grammar. 

50 Skip forms completely for invariable words and fixed expressions. 

514. Usage: collocations, governed prepositions, confusions, register and 

52 countability where relevant. 

535. Origin only where you know it: 1-3 sentences for a borrowing, one compact line 

54 for a native word. Where you do not, leave it out — an origin reasoned out from 

55 the parts of a word reads exactly like one you know, and teaches a fiction. 

566. Give 2-4 short everyday examples with translations. 

57For a set expression, also explain what its parts contribute and why the whole 

58means what it does. With supplied context, lead with the sense used there but 

59keep all other senses below it. The heading, the translations and the examples are 

60all about one and the same wording. Where a submission means what it means because

227def build_prompt( 

228 language: Language, 

229 word: str, 

230 target_lang: str, 

231 *, 

232 context: str = "", 

233 unit_intent: bool = False, 

234) -> str: 

235 """Build the prompt for the branch the request is in, or for deciding it.""" 

236 request = ( 

237 f'Make a card for this selected unit: "{word}"\nContext: "{context}"' 

238 if unit_intent and context 

239 else f'Make a card for this selected unit: "{word}"' 

240 if unit_intent 

241 else f'Analyse this submitted text: "{word}"' 

242 ) 

243 template = _SELECTED_PROMPT if unit_intent else _OPEN_PROMPT 

244 return template.format( 

245 source_lang=language.name, 

246 target_lang=target_lang, 

247 source_hints=language.prompt_hints or "", 

248 request=request, 

249 context_field=', "context_sense": <zero-based index>' if context else "", 

250 context_rule=_CONTEXT_RULE if context else _NO_CONTEXT_RULE, 

251 )

292def parse_attestation(raw: str) -> Verdict | None: 

293 """Read the judgement, which is one bare JSON object and nothing else.""" 

294 start = raw.find("{") 

295 if start < 0: 

296 return None 

297 try: 

298 value, _consumed = json.JSONDecoder().raw_decode(raw[start:]) 

299 except ValueError: 

300 return None 

301 used = value.get("used") if isinstance(value, dict) else None 

302 return Verdict(used) if isinstance(used, bool) else None

305def extract_answer( # noqa: PLR0913 - the whole request the answer is read against. 

306 raw: str, 

307 submitted: str, 

308 language: Language, 

309 *, 

310 unit_intent: bool = False, 

311 context: str = "", 

312 target: str = DEFAULT_TARGET_LANGUAGE, 

313) -> ParsedAnswer | None: 

314 """Extract and validate the hidden answer, returning None when unusable.""" 

315 if len(raw) > MAX_COMPLETE_ANSWER_CHARS: 

316 logger.warning( 

317 "oversized answer for %s/%r: %s characters exceeds the %s-character bound", 

318 language.code, 

319 submitted, 

320 len(raw), 

321 MAX_COMPLETE_ANSWER_CHARS, 

322 ) 

323 return None 

324 delimiter_at = raw.find(CARD_DELIMITER) 

325 if delimiter_at < 0: 

326 logger.warning( 

327 "no answer block for %s/%r: the answer never wrote the delimiter", 

328 language.code, 

329 submitted, 

330 ) 

331 return None 

332 payload = raw[delimiter_at + len(CARD_DELIMITER) :] 

333 try: 

334 return parse_answer_payload( 

335 payload, 

336 submitted, 

337 language, 

338 unit_intent=unit_intent, 

339 context=context, 

340 target=target, 

341 ) 

342 except CardParseError as exc: 

343 logger.warning( 

344 "unusable answer block for %s/%r: %s; payload %s", 

345 language.code, 

346 submitted, 

347 exc, 

348 _logged(payload), 

349 ) 

350 return None

353def _logged(payload: str) -> str: 

354 payload = payload.strip() 

355 if len(payload) <= PAYLOAD_LOG_LIMIT: 

356 return repr(payload) 

357 return f"{payload[:PAYLOAD_LOG_LIMIT]!r} … ({len(payload)} chars)"
