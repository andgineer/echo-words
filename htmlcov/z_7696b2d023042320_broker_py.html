26@lru_cache(maxsize=1) 

27def llmbroker() -> ModuleType: 

28 """The ``llmbroker`` module, imported at call time.""" 

29 # Imported here rather than at module level so a missing or broken install is a 

30 # clear config error on the first word instead of a crash before the app serves. 

31 try: 

32 import llmbroker as module # noqa: PLC0415 

33 except ImportError as exc: 

34 raise BackendError(f"llmbroker is not usable: {exc}") from exc 

35 return module

38def paid_alias(language: Language, settings: Settings) -> str: 

39 """The paid model this language reaches, or empty when the paid step is switched off.""" 

40 # An empty ECHOWORDS_API_MODEL switches the second step off app-wide: a language's 

41 # own api_model chooses which paid model answers, never whether one is paid for. 

42 if not settings.api_model: 

43 return "" 

44 return language.api_model or settings.api_model

46 

47def paid_aliases(languages: dict[str, Language], settings: Settings) -> list[str]: 

48 """Every paid alias a language can reach; llmbroker takes them at construction.""" 

49 aliases: list[str] = [] 

50 for language in languages.values(): 

51 alias = paid_alias(language, settings) 

52 if alias and alias not in aliases: 

53 aliases.append(alias) 

54 return aliases
